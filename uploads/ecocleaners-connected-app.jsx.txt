function stageStatus(stage) {
  if (stage >= 4) return "completed";
  if (stage === 0) return "pending";
  return "progress";
}

function StatusBadge({ stage }) {
  const map = {
    pending: { bg: "#FEF3C7", fg: C.warning, label: "Awaiting Pickup" },
    progress: { bg: "#DBEAFE", fg: C.info, label: STAGES[stage - 1]?.label || "In Progress" },
    completed: { bg: "#DCFCE7", fg: C.success, label: "Delivered" },
  };
  const s = map[stageStatus(stage)];
  return (
    <span className="px-2.5 py-1 rounded-full text-xs font-semibold whitespace-nowrap" style={{ background: s.bg, color: s.fg }}>
      {s.label}
    </span>
  );
}

function Card({ children, className = "", onClick }) {
  return (
    <div
      onClick={onClick}
      className={`bg-white rounded-2xl border border-gray-100 ${className}`}
      style={{ boxShadow: "0 1px 3px rgba(0,0,0,0.04), 0 1px 2px rgba(0,0,0,0.03)" }}
    >
      {children}
    </div>
  );
}

function PrimaryButton({ children, onClick, className = "", disabled, color }) {
  return (
    <button
      onClick={onClick}
      disabled={disabled}
      className={`w-full rounded-2xl font-semibold text-white transition-all active:scale-[0.98] disabled:opacity-40 ${className}`}
      style={{ background: disabled ? "#D1D5DB" : color || C.primary, minHeight: 52 }}
    >
      {children}
    </button>
  );
}

function SecondaryButton({ children, onClick, className = "" }) {
  return (
    <button
      onClick={onClick}
      className={`w-full rounded-2xl font-semibold transition-all active:scale-[0.98] ${className}`}
      style={{ background: C.section, color: C.text, minHeight: 52 }}
    >
      {children}
    </button>
  );
}

function MiniTimeline({ stage }) {
  return (
    <div className="flex items-center">
      {STAGES.map((s, i) => (
        <React.Fragment key={s.key}>
          <div
            className="w-6 h-6 rounded-full flex items-center justify-center flex-shrink-0"
            style={{ background: i < stage ? C.primary : C.section, transition: "background 0.3s" }}
          >
            <s.icon size={12} color={i < stage ? "#fff" : C.sub} />
          </div>
          {i < STAGES.length - 1 && (
            <div className="flex-1 h-1 mx-1 rounded-full" style={{ background: i < stage - 1 ? C.primary : C.section, transition: "background 0.3s" }} />
          )}
        </React.Fragment>
      ))}
    </div>
  );
}

function CustomerBook({ onConfirm }) {
  const [step, setStep] = useState(0);
  const [serviceId, setServiceId] = useState(null);
  const [slot, setSlot] = useState(null);
  const steps = ["Service", "Time", "Confirm"];
  const canNext = (step === 0 && serviceId) || (step === 1 && slot) || step === 2;
  const service = services.find((s) => s.id === serviceId);

  if (step === 3) {
    return (
      <div className="flex flex-col items-center justify-center px-8 pt-24 text-center">
        <div className="w-20 h-20 rounded-full flex items-center justify-center mb-5" style={{ background: "#D1FAE5" }}>
          <CheckCircle2 size={40} color={C.primary} />
        </div>
        <h1 className="text-xl font-bold" style={{ color: C.text }}>Pickup booked!</h1>
        <p className="text-sm mt-2" style={{ color: C.sub }}>
          We'll collect your laundry during {slot}. This order is now visible to dispatch and your assigned driver.
        </p>
        <PrimaryButton onClick={() => onConfirm(null)} className="py-4 mt-8">Back to Home</PrimaryButton>
      </div>
    );
  }

  return (
    <div className="pb-4">
      <div className="px-5 pt-6 pb-2"><h1 className="text-2xl font-bold" style={{ color: C.text }}>Book a Pickup</h1></div>
      <div className="px-5 flex items-center gap-2 mt-1 mb-4">
        {steps.map((label, i) => (
          <div key={label} className="h-1.5 rounded-full flex-1" style={{ background: i <= step ? C.primary : C.section, transition: "background 0.3s" }} />
        ))}
      </div>
      <p className="px-5 text-xs font-semibold mb-3" style={{ color: C.sub }}>STEP {step + 1} OF 3 · {steps[step].toUpperCase()}</p>

      {step === 0 && (
        <div className="px-5 flex flex-col gap-3">
          {services.map((s) => (
            <Card key={s.id} className="p-4 flex items-center gap-3 cursor-pointer" onClick={() => setServiceId(s.id)}>
              <div className="w-11 h-11 rounded-xl flex items-center justify-center flex-shrink-0" style={{ background: serviceId === s.id ? C.primary : "#D1FAE5" }}>
                <s.icon size={20} color={serviceId === s.id ? "#fff" : C.primary} />
              </div>
              <div className="flex-1">
                <p className="text-sm font-semibold" style={{ color: C.text }}>{s.name}</p>
                <p className="text-xs" style={{ color: C.sub }}>{s.desc}</p>
              </div>
              <p className="text-xs font-semibold" style={{ color: C.text }}>{s.price}</p>
            </Card>
          ))}
        </div>
      )}

      {step === 1 && (
        <div className="px-5">
          <div className="flex items-center gap-2 mb-4 text-sm" style={{ color: C.text }}>
            <Calendar size={16} color={C.primary} /> Today, Jul 3
          </div>
          <div className="grid grid-cols-2 gap-3">
            {timeSlots.map((t) => (
              <Card key={t} className="p-3 text-center cursor-pointer" onClick={() => setSlot(t)}>
                <p className="text-sm font-semibold" style={{ color: slot === t ? C.primary : C.text }}>{t}</p>
              </Card>
            ))}
          </div>
        </div>
      )}

      {step === 2 && (
        <div className="px-5 flex flex-col gap-3">
          <Card className="p-4">
            <div className="flex justify-between text-sm mb-2"><span style={{ color: C.sub }}>Service</span><span className="font-semibold" style={{ color: C.text }}>{service?.name}</span></div>
            <div className="flex justify-between text-sm mb-2"><span style={{ color: C.sub }}>Pickup window</span><span className="font-semibold" style={{ color: C.text }}>{slot}</span></div>
            <div className="flex justify-between text-sm"><span style={{ color: C.sub }}>Address</span><span className="font-semibold" style={{ color: C.text }}>{CUSTOMER_ADDRESS}</span></div>
          </Card>
          <Card className="p-4 flex items-center gap-3">
            <CreditCard size={18} color={C.blue} />
            <p className="text-sm flex-1" style={{ color: C.text }}>Pay via M-PESA on delivery</p>
          </Card>
        </div>
      )}

      <div className="px-5 mt-6 flex gap-3">
        {step > 0 && <SecondaryButton onClick={() => setStep(step - 1)} className="py-3.5 flex-1">Back</SecondaryButton>}
        <PrimaryButton
          disabled={!canNext}
          onClick={() => {
            if (step === 2) onConfirm({ service, slot });
            setStep(step + 1);
          }}
          className="py-3.5 flex-1"
        >
          {step === 2 ? "Confirm Booking" : "Continue"}
        </PrimaryButton>
      </div>
    </div>
  );
}

function CustomerApp({ orders, addOrder }) {
  const [tab, setTab] = useState("home");
  const [selected, setSelected] = useState(null);

  const openOrder = (o) => setSelected(o);
  const closeOrder = () => setSelected(null);
  const goBook = () => setTab("book");

  const handleConfirm = (booking) => {
    if (booking) {
      const id = `EC-${orderSeq++}`;
      addOrder({
        id, customer: CUSTOMER_NAME, phone: CUSTOMER_PHONE, address: CUSTOMER_ADDRESS,
        serviceName: booking.service.name, serviceIcon: booking.service.icon, items: Math.floor(Math.random() * 6) + 4,
        amount: booking.service.amount, window: booking.slot, date: "Today", stage: 0,
      });
    } else {
      setTab("home");
    }
  };

  let screen;
  if (selected) screen = <CustomerOrderDetail order={selected} onBack={closeOrder} />;
  else if (tab === "home") screen = <CustomerHome orders={orders} goBook={goBook} openOrder={openOrder} />;
  else if (tab === "orders") screen = <CustomerOrders orders={orders} openOrder={openOrder} />;
  else if (tab === "book") screen = <CustomerBook onConfirm={handleConfirm} />;
  else if (tab === "notifications") screen = <CustomerNotifications />;
  else if (tab === "profile") screen = <CustomerProfile />;

  const items = [
    { key: "home", label: "Home", icon: Home },
    { key: "orders", label: "Orders", icon: Package },
    { key: "book", label: "Book", icon: Plus, center: true },
    { key: "notifications", label: "Alerts", icon: Bell },
    { key: "profile", label: "Profile", icon: User },
  ];

  return (
    <div className="relative w-[390px] h-[780px] bg-white rounded-[40px] overflow-hidden shadow-2xl border-[6px]" style={{ borderColor: "#111827", fontFamily: FONT }}>
      <div className="h-11 flex items-center justify-between px-6 text-xs font-semibold" style={{ color: C.text }}>
        <span>9:41</span><span>EcoCleaners</span>
      </div>
      <div className="overflow-y-auto" style={{ height: "calc(100% - 44px - 78px)" }}>{screen}</div>
      <div className="absolute bottom-0 left-0 right-0 flex items-end justify-around px-2 pb-2 pt-2" style={{ background: "#fff", borderTop: "1px solid #F3F4F6" }}>
        {items.map((it) =>
          it.center ? (
            <button key={it.key} onClick={() => { setSelected(null); setTab(it.key); }} className="flex flex-col items-center -mt-6">
              <div className="w-14 h-14 rounded-full flex items-center justify-center shadow-lg" style={{ background: C.primary }}>
                <Plus size={26} color="#fff" />
              </div>
            </button>
          ) : (
            <button key={it.key} onClick={() => { setSelected(null); setTab(it.key); }} className="flex flex-col items-center gap-1 py-1 px-3" style={{ minHeight: 44, minWidth: 44 }}>
              <it.icon size={20} color={tab === it.key && !selected ? C.primary : "#9CA3AF"} />
              <span className="text-[10px] font-medium" style={{ color: tab === it.key && !selected ? C.primary : "#9CA3AF" }}>{it.label}</span>
            </button>
          )
        )}
      </div>
    </div>
  );
}

function JobBadge({ type }) {
  const map = { pickup: { bg: "#DBEAFE", fg: C.blue, label: "PICKUP" }, delivery: { bg: "#D1FAE5", fg: C.primary, label: "DELIVERY" } };
  const s = map[type];
  return <span className="px-2.5 py-1 rounded-full text-[11px] font-bold tracking-wide" style={{ background: s.bg, color: s.fg }}>{s.label}</span>;
}

function BigButton({ children, onClick, color = C.primary }) {
  return (
    <button onClick={onClick} className="w-full rounded-2xl font-bold text-white text-base transition-transform active:scale-[0.98]" style={{ background: color, minHeight: 56 }}>
      {children}
    </button>
  );
}

function driverJobsFrom(orders) {
  const pickups = orders.filter((o) => o.stage === 0).map((o) => ({ ...o, type: "pickup" }));
  const deliveries = orders.filter((o) => o.stage === 3).map((o) => ({ ...o, type: "delivery" }));
  const done = orders.filter((o) => o.stage === 4).map((o) => ({ ...o, type: "delivery" }));
  return { pending: [...pickups, ...deliveries], done };
}

function DriverJobs({ orders, openJob }) {
  const { pending, done } = driverJobsFrom(orders);
  return (
    <div className="pb-4">
      <div className="px-5 pt-6 pb-2">
        <p className="text-sm" style={{ color: C.sub }}>Today, Jul 3</p>
        <h1 className="text-2xl font-bold" style={{ color: C.text, letterSpacing: "-0.02em" }}>{pending.length} jobs left</h1>
      </div>
      <div className="px-5 mt-3 flex flex-col gap-3">
        {pending.map((j) => (
          <div key={j.id + j.type} onClick={() => openJob(j)} className="rounded-2xl p-4 cursor-pointer bg-white border border-gray-100" style={{ boxShadow: "0 1px 3px rgba(0,0,0,0.05)" }}>
            <div className="flex items-center justify-between mb-2"><JobBadge type={j.type} /><span className="text-xs font-semibold" style={{ color: C.warning }}>{j.window}</span></div>
            <p className="text-base font-bold" style={{ color: C.text }}>{j.customer}</p>
            <div className="flex items-center gap-1.5 mt-1"><MapPin size={13} color={C.sub} /><p className="text-xs" style={{ color: C.sub }}>{j.address}</p></div>
          </div>
        ))}
        {pending.length === 0 && <p className="text-sm text-center py-8" style={{ color: C.sub }}>No jobs right now. New pickups appear here automatically.</p>}
      </div>
      {done.length > 0 && (
        <div className="px-5 mt-6">
          <p className="text-xs font-bold uppercase tracking-wide mb-3" style={{ color: C.sub }}>Completed</p>
          <div className="flex flex-col gap-3">
            {done.map((j) => (
              <div key={j.id} className="rounded-2xl p-4 bg-white border border-gray-100 opacity-60">
                <div className="flex items-center justify-between">
                  <div className="flex items-center gap-2"><CheckCircle2 size={16} color={C.success} /><p className="text-sm font-semibold" style={{ color: C.text }}>{j.customer}</p></div>
                  <span className="text-xs" style={{ color: C.sub }}>{j.window}</span>
                </div>
              </div>
            ))}
          </div>
        </div>
      )}
    </div>
  );
}

function DriverApp({ orders, advanceOrder }) {
  const [tab, setTab] = useState("jobs");
  const [selectedJob, setSelectedJob] = useState(null);

  let screen;
  if (selectedJob) screen = <DriverJobDetail job={selectedJob} onBack={() => setSelectedJob(null)} onAdvance={advanceOrder} />;
  else if (tab === "jobs") screen = <DriverJobs orders={orders} openJob={setSelectedJob} />;
  else if (tab === "map") screen = <DriverMap orders={orders} />;
  else if (tab === "profile") screen = <DriverProfile orders={orders} />;

  const items = [{ key: "jobs", label: "Jobs", icon: Briefcase }, { key: "map", label: "Map", icon: MapIcon }, { key: "profile", label: "Profile", icon: User }];

  return (
    <div className="relative w-[390px] h-[780px] bg-white rounded-[40px] overflow-hidden shadow-2xl border-[6px] flex" style={{ borderColor: "#111827", fontFamily: FONT }}>
      {!selectedJob && (
        <div className="w-20 h-full flex flex-col items-center gap-2 pt-6 flex-shrink-0" style={{ background: "#fff", borderRight: `1px solid ${C.section}` }}>
          {items.map((it) => {
            const isActive = tab === it.key;
            return (
              <button key={it.key} onClick={() => setTab(it.key)} className="flex flex-col items-center gap-1 py-3 rounded-2xl w-16" style={{ minHeight: 44, background: isActive ? "#D1FAE5" : "transparent" }}>
                <it.icon size={20} color={isActive ? C.primary : C.sub} />
                <span className="text-[10px] font-semibold" style={{ color: isActive ? C.primary : C.sub }}>{it.label}</span>
              </button>
            );
          })}
        </div>
      )}
      <div className="flex-1 h-full overflow-y-auto" style={{ background: "#fff" }}>{screen}</div>
    </div>
  );
}

export default function EcoCleanersConnectedDemo() {
  const [orders, setOrders] = useState(seedOrders());
  const [role, setRole] = useState("customer");

  const addOrder = (order) => setOrders((prev) => [order, ...prev]);

  const advanceOrder = (id, type) => {
    setOrders((prev) => prev.map((o) => {
      if (o.id !== id) return o;
      if (type === "pickup") return { ...o, stage: 1 };
      if (type === "delivery") return { ...o, stage: 4 };
      return o;
    }));
  };

  const simulateWashFold = () => {
    setOrders((prev) => prev.map((o) => (o.stage === 1 || o.stage === 2) ? { ...o, stage: o.stage + 1 } : o));
  };

  return (
    <div className="min-h-screen flex flex-col items-center py-6" style={{ background: "#E5E7EB", fontFamily: FONT }}>
      {/* Demo-only control — not part of any real product surface */}
      <div className="flex items-center gap-1 bg-white rounded-full p-1 mb-6 shadow-sm border border-gray-200">
        {[
          { key: "customer", label: "Customer App" },
          { key: "admin", label: "Admin Dashboard" },
          { key: "driver", label: "Driver App" },
        ].map((r) => (
          <button
            key={r.key}
            onClick={() => setRole(r.key)}
            className="px-4 py-2 rounded-full text-xs font-semibold transition-colors"
            style={{ background: role === r.key ? C.text : "transparent", color: role === r.key ? "#fff" : C.sub }}
          >
            {r.label}
          </button>
        ))}
      </div>

      {role === "customer" && <CustomerApp orders={orders} addOrder={addOrder} />}
      {role === "admin" && (
        <div className="w-[1100px] h-[720px] rounded-2xl overflow-hidden shadow-2xl">
          <AdminApp orders={orders} onSimulate={simulateWashFold} />
        </div>
      )}
      {role === "driver" && <DriverApp orders={orders} advanceOrder={advanceOrder} />}
    </div>
  );
}
