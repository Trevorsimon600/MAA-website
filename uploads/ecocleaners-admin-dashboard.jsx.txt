// ---------- Reusable primitives ----------
function StatusBadge({ status }) {
  const map = {
    completed: { bg: "#DCFCE7", fg: C.success, label: "Completed" },
    progress: { bg: "#FEF3C7", fg: C.warning, label: "In Progress" },
    tracking: { bg: "#DBEAFE", fg: C.info, label: "Tracking" },
    cancelled: { bg: "#FEE2E2", fg: C.error, label: "Cancelled" },
  };
  const s = map[status] || map.completed;
  return (
    <span className="px-2.5 py-1 rounded-full text-xs font-semibold whitespace-nowrap" style={{ background: s.bg, color: s.fg }}>
      {s.label}
    </span>
  );
}

function Card({ children, className = "" }) {
  return (
    <div
      className={`bg-white rounded-2xl border border-gray-100 ${className}`}
      style={{ boxShadow: "0 1px 3px rgba(0,0,0,0.04), 0 1px 2px rgba(0,0,0,0.03)" }}
    >
      {children}
    </div>
  );
}

}
